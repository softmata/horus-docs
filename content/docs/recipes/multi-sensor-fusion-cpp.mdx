## Multi-Sensor Fusion (C++)

Fuse IMU orientation with wheel odometry for more accurate localization.

### Code

```cpp
#include <horus/horus.hpp>
#include <cmath>
using namespace horus::literals;

class SensorFusion : public horus::Node {
public:
    SensorFusion() : Node("sensor_fusion") {
        imu_sub_  = subscribe<horus::msg::Imu>("imu.data");
        odom_sub_ = subscribe<horus::msg::Odometry>("odom");
        fused_pub_ = advertise<horus::msg::Odometry>("fused.pose");
    }

    void tick() override {
        // Read both sensors
        auto imu  = imu_sub_->recv();
        auto odom = odom_sub_->recv();

        // Update from odometry (predict)
        if (odom) {
            x_ = odom->get()->pose.x;
            y_ = odom->get()->pose.y;
            // Blend odometry heading with current estimate
            double odom_theta = odom->get()->pose.theta;
            theta_ = alpha_ * theta_ + (1.0 - alpha_) * odom_theta;
        }

        // Correct from IMU (gyro-based heading)
        if (imu) {
            double gz = imu->get()->angular_velocity[2];
            double imu_theta = theta_ + gz * 0.01;  // integrate gyro
            theta_ = alpha_ * imu_theta + (1.0 - alpha_) * theta_;
        }

        // Publish fused estimate
        horus::msg::Odometry fused{};
        fused.pose.x = x_;
        fused.pose.y = y_;
        fused.pose.theta = theta_;
        fused_pub_->send(fused);
    }

private:
    horus::Subscriber<horus::msg::Imu>*      imu_sub_;
    horus::Subscriber<horus::msg::Odometry>* odom_sub_;
    horus::Publisher<horus::msg::Odometry>*  fused_pub_;
    double x_ = 0, y_ = 0, theta_ = 0;
    double alpha_ = 0.95;  // IMU trust factor
};

int main() {
    horus::Scheduler sched;
    sched.tick_rate(100_hz);
    SensorFusion fusion;
    sched.add(fusion).order(20).build();
    sched.spin();
}
```
