## Servo Controller (C++)

Control a servo motor with position commands. Use `horus::Params` for live gain tuning.

### Code

```cpp
#include <horus/horus.hpp>
#include <cmath>
using namespace horus::literals;

class ServoController : public horus::Node {
public:
    ServoController() : Node("servo_controller") {
        target_sub_ = subscribe<horus::msg::CmdVel>("servo.target");
        motor_pub_  = advertise<horus::msg::ServoCommand>("servo.cmd");
    }

    void tick() override {
        auto target = target_sub_->recv();
        if (!target) return;

        double kp = params_.get<double>("servo_kp", 2.0);
        double target_pos = target->get()->linear;
        double error = target_pos - current_pos_;
        double effort = std::clamp(kp * error, -1.0, 1.0);

        // Simulate servo motion
        current_pos_ += effort * 0.01;

        horus::msg::ServoCommand cmd{};
        cmd.target_position = static_cast<float>(target_pos);
        cmd.target_velocity = static_cast<float>(effort);
        cmd.max_force = 5.0f;
        motor_pub_->send(cmd);
    }

    void set_params(horus::Params& p) { params_ = std::move(p); }

private:
    horus::Subscriber<horus::msg::CmdVel>*     target_sub_;
    horus::Publisher<horus::msg::ServoCommand>* motor_pub_;
    horus::Params params_;
    double current_pos_ = 0;
};

int main() {
    horus::Scheduler sched;
    sched.tick_rate(100_hz);

    horus::Params params;
    params.set("servo_kp", 2.0);

    ServoController servo;
    servo.set_params(params);
    sched.add(servo).order(10).build();
    sched.spin();
}
```
