## IMU Reader (C++)

Read accelerometer + gyroscope data, fuse with a complementary filter, publish orientation.

### Code

```cpp
#include <horus/horus.hpp>
#include <cmath>
using namespace horus::literals;

class ImuReader : public horus::Node {
public:
    ImuReader() : Node("imu_reader") {
        imu_sub_  = subscribe<horus::msg::Imu>("imu.raw");
        pose_pub_ = advertise<horus::msg::Pose2D>("imu.orientation");
    }

    void tick() override {
        auto data = imu_sub_->recv();
        if (!data) return;

        const auto* imu = data->get();
        double ax = imu->linear_acceleration[0];
        double ay = imu->linear_acceleration[1];
        double az = imu->linear_acceleration[2];
        double gz = imu->angular_velocity[2];

        // Accelerometer angle (pitch from gravity)
        double accel_angle = std::atan2(ay, az);

        // Complementary filter: 98% gyro, 2% accelerometer
        double dt = 0.01;
        angle_ = 0.98 * (angle_ + gz * dt) + 0.02 * accel_angle;

        horus::msg::Pose2D pose{};
        pose.theta = angle_;
        pose_pub_->send(pose);
    }

private:
    horus::Subscriber<horus::msg::Imu>*   imu_sub_;
    horus::Publisher<horus::msg::Pose2D>*  pose_pub_;
    double angle_ = 0;
};

int main() {
    horus::Scheduler sched;
    sched.tick_rate(100_hz);
    ImuReader imu;
    sched.add(imu).order(5).build();
    sched.spin();
}
```
