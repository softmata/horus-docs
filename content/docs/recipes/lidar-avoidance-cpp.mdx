## LiDAR Obstacle Avoidance (C++)

Read 360-degree LiDAR scans, find closest obstacle, steer away or stop.

### Code

```cpp
#include <horus/horus.hpp>
#include <algorithm>
#include <cmath>
using namespace horus::literals;

class LidarAvoidance : public horus::Node {
public:
    LidarAvoidance(float safe_dist = 0.5f, float max_speed = 0.3f)
        : Node("lidar_avoidance"), safe_dist_(safe_dist), max_speed_(max_speed)
    {
        scan_sub_ = subscribe<horus::msg::LaserScan>("lidar.scan");
        cmd_pub_  = advertise<horus::msg::CmdVel>("cmd_vel");
    }

    void tick() override {
        auto scan = scan_sub_->recv();
        if (!scan) return;

        const auto* s = scan->get();

        // Find minimum range and its angle
        float min_range = 999.0f;
        int min_idx = 0;
        for (int i = 0; i < 360; i++) {
            if (s->ranges[i] > 0.01f && s->ranges[i] < min_range) {
                min_range = s->ranges[i];
                min_idx = i;
            }
        }

        horus::msg::CmdVel cmd{};
        if (min_range < safe_dist_) {
            // Obstacle close — turn away
            cmd.linear = 0.0f;
            cmd.angular = (min_idx < 180) ? -0.5f : 0.5f;
            horus::log::warn("lidar_avoidance", "Obstacle detected, turning");
        } else {
            // Clear — go forward
            cmd.linear = max_speed_;
            cmd.angular = 0.0f;
        }
        cmd_pub_->send(cmd);
    }

    void enter_safe_state() override {
        horus::msg::CmdVel stop{};
        cmd_pub_->send(stop);
    }

private:
    horus::Subscriber<horus::msg::LaserScan>* scan_sub_;
    horus::Publisher<horus::msg::CmdVel>*     cmd_pub_;
    float safe_dist_, max_speed_;
};

int main() {
    horus::Scheduler sched;
    sched.tick_rate(10_hz);
    LidarAvoidance avoid(0.5f, 0.3f);
    sched.add(avoid).order(10).build();
    sched.spin();
}
```
