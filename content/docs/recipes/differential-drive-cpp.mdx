## Differential Drive (C++)

Control a two-wheeled robot: convert CmdVel (linear + angular) into left/right wheel velocities, with odometry feedback.

### Code

```cpp
#include <horus/horus.hpp>
#include <cmath>
using namespace horus::literals;

class DiffDrive : public horus::Node {
public:
    DiffDrive(double wheel_base, double wheel_radius)
        : Node("diff_drive"), wheel_base_(wheel_base), wheel_radius_(wheel_radius)
    {
        cmd_sub_   = subscribe<horus::msg::CmdVel>("cmd_vel");
        odom_pub_  = advertise<horus::msg::Odometry>("odom");
        motor_pub_ = advertise<horus::msg::CmdVel>("motor.wheels");
    }

    void tick() override {
        auto cmd = cmd_sub_->recv();
        if (!cmd) return;

        double v = cmd->get()->linear;
        double w = cmd->get()->angular;

        // Inverse kinematics: CmdVel -> wheel velocities
        double v_left  = (v - w * wheel_base_ / 2.0) / wheel_radius_;
        double v_right = (v + w * wheel_base_ / 2.0) / wheel_radius_;

        horus::msg::CmdVel wheels{};
        wheels.linear  = static_cast<float>(v_left);
        wheels.angular = static_cast<float>(v_right);
        motor_pub_->send(wheels);

        // Forward kinematics: update odometry
        double dt = 0.01;  // 100Hz
        x_     += v * std::cos(theta_) * dt;
        y_     += v * std::sin(theta_) * dt;
        theta_ += w * dt;

        horus::msg::Odometry odom{};
        odom.pose.x = x_;
        odom.pose.y = y_;
        odom.pose.theta = theta_;
        odom_pub_->send(odom);
    }

    void enter_safe_state() override {
        horus::msg::CmdVel stop{};
        motor_pub_->send(stop);
    }

private:
    horus::Subscriber<horus::msg::CmdVel>* cmd_sub_;
    horus::Publisher<horus::msg::Odometry>* odom_pub_;
    horus::Publisher<horus::msg::CmdVel>*  motor_pub_;
    double wheel_base_, wheel_radius_;
    double x_ = 0, y_ = 0, theta_ = 0;
};

int main() {
    horus::Scheduler sched;
    sched.tick_rate(100_hz);

    DiffDrive drive(0.3, 0.05);  // 30cm wheelbase, 5cm wheel radius
    sched.add(drive).order(10).build();
    sched.spin();
}
```
